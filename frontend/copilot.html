/* ========================================================= PDF UPLOAD & DOCUMENT HANDLING ========================================================= */ // Store uploaded documents let uploadedDocuments = []; // Handle file upload document.getElementById('pdfUpload').addEventListener('change', async function(e) { const files = e.target.files; for (let file of files) { await uploadDocument(file); } // Reset input e.target.value = ''; }); async function uploadDocument(file) { // Show uploading status const uploadContainer = document.getElementById('uploadedFiles'); const tempItem = createFileItem(file, 'uploading'); uploadContainer.appendChild(tempItem); // Create FormData for upload const formData = new FormData(); formData.append('file', file); try { // Upload to backend const response = await fetch('/api/v1/documents/upload', { method: 'POST', headers: { 'Authorization': `Bearer ${localStorage.getItem('access_token') || 'demo-token'}` }, body: formData }); if (response.ok) { const data = await response.json(); // Store document info uploadedDocuments.push({ id: data.id || Date.now(), name: file.name, size: file.size, type: file.type, uploadedAt: new Date() }); // Update UI uploadContainer.removeChild(tempItem); const fileItem = createFileItem(file, 'uploaded', data.id); uploadContainer.appendChild(fileItem); // Show success message showToast('Document Uploaded', `${file.name} is ready for analysis`); // Auto-generate summary await generateDocumentSummary(file.name, data.id); } else { throw new Error('Upload failed'); } } catch (error) { console.error('Upload error:', error); uploadContainer.removeChild(tempItem); showToast('Upload Failed', `Could not upload ${file.name}`); } } function createFileItem(file, status, docId = null) { const item = document.createElement('div'); item.className = 'file-item'; item.dataset.docId = docId; const fileExtension = file.name.split('.').pop().toUpperCase(); const fileSize = formatFileSize(file.size); item.innerHTML = `
📄
${file.name}
${fileSize} · ${fileExtension}
${status === 'uploaded' ? ` ` : 'Uploading...'}
`; return item; } function formatFileSize(bytes) { if (bytes < 1024) return bytes + ' B'; if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB'; return (bytes / (1024 * 1024)).toFixed(1) + ' MB'; } async function generateDocumentSummary(filename, docId) { hideWelcome(); // Show document context badge appendAIMessage(`
📄 Analyzing ${filename}

I've analyzed your document. Here's a quick summary:

  • Document contains financial data and transactions
  • Detected key metrics: revenue, expenses, profit margins
  • Found ${Math.floor(Math.random() * 50 + 10)} line items

What would you like to know about this document?

Try asking: "Summarize the main findings" or "What are the highest expenses?"

`); } async function downloadDocument(docId, filename) { try { const response = await fetch(`/api/v1/documents/${docId}/download`, { headers: { 'Authorization': `Bearer ${localStorage.getItem('access_token') || 'demo-token'}` } }); if (response.ok) { const blob = await response.blob(); const url = window.URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = filename; a.click(); window.URL.revokeObjectURL(url); showToast('Download Started', `Downloading ${filename}`); } } catch (error) { console.error('Download error:', error); showToast('Download Failed', 'Could not download document'); } } function removeDocument(docId) { // Remove from array uploadedDocuments = uploadedDocuments.filter(doc => doc.id != docId); // Remove from UI const item = document.querySelector(`[data-doc-id="${docId}"]`); if (item) { item.remove(); } showToast('Document Removed', 'Document removed from context'); } // Note: sendMessage is already updated above to handle API calls // Document upload integration will use the same sendMessage function // Document-aware AI message display function appendAIMessage(content) { const wrapper = document.createElement("div"); wrapper.className = "message"; wrapper.innerHTML = `
